    markdown2pdf \
          -N \

          --variable version=1.4 \
          efProjectUpdate.md \
          --xetex \
          --toc \
          --output=/home/ef/Documents/efProject.pdf

### insert latex
    - $ or \(
    - inline equation: `$my latex equation$`
    - inline equation: `\(my latex equation\)`

    - $$ or \[
    - display mode: `$$my equation$$`
    - display mode: `\[my equation\]`

# from markdown to pdf
    ```
    #!/bin/bash
    ProjectHome="/home/xx/Documents/project/Text"

    cd "$ProjectHome"
    echo `pwd`
    echo "executing..."

    # Use pandoc to create a pdf from the markdown + tex file
    markdown2pdf \
                -N \
                --template=myProject.tex \
                --variable version=1.4 \
                myProjectUpdate.md \
                --xetex \
                --toc \
                --bibliography=Project.bib \
                --csl=ieee.csl \
                --output="$ProjectHome"/myProject.pdf
    ```
