### change vim cursor (insert into: ~/.vimrc.local)
    " URXVT has the ability to use  
    " a fancy cursor for insert mode

    if exists('$TMUX')
        let &t_SI = "\<Esc>Ptmux;\<Esc>\e[6 q\<Esc>\\"
        let &t_EI = "\<Esc>Ptmux;\<Esc>\e[2 q\<Esc>\\"
    else
        " set (this linking) i-beam 
        let &t_SI = "\e[6 q"
        " set (kon-blinking) block
        let &t_EI = "\e[2 q"
    endif

    c rxvt-unicode/

        export PKG_CONFIG_PATH=$HOME/mylibs/lib/pkgconfig:$PKG_CONFIG_PATH 
    ### Configure: static link libptytty
        LDFLAGS="-L$HOME/mylibs/lib -lptytty" CPPFLAGS="-I$HOME/mylibs/include" ./configure --prefix=$HOME/.my-urxvt
    make
    make install

    - Note: may need autoconf > 2.71

        Assume `libptytty` source is at: ~/libptytty

        cmake --build .
        cmake --install .


### utf8 symbol glyphs
    if symbols not displaying, set a fallback font in .Xdefaults!
    e.g. Monospace is the fallback here!
    `URxvt*font: xft:Monaco:size=7.5, xft:Monospace:size=8`

### autostart tmux
    urxvt -e tmux

### copy text
    simply select it. Then
    1. goto new window and hit middle mouse button
    or
    2. hit SHIFT-INSERT

### text selection
    double-click word to select it
    treble click to select line
