# bash completion for my 'coa' alias: activating conda environments
_coa() {
    local cur=${COMP_WORDS[COMP_CWORD]}
    local envs=$(awk -F'/' '{print $NF}' $HOME/.conda/environments.txt)
    COMPREPLY=($(compgen -W "${envs}" -- ${cur}))
    __ltrim_colon_completions "$cur"
}
complete -F _coa coa
