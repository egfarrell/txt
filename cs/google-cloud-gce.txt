### Attach disk
    lsblk
    lsblk -f

    - `bio-disk "/dev/sdb"`
        sudo mkdir /bio-tarballs
        sudo chown -R ef /bio-tarballs/
        sudo mount /dev/sdb /bio-tarballs
        > - UNMOUNT
            sudo umount -d /dev/sdb

    - `rosetta-run-disk "/dev/sdb"`
        sudo mkdir /rosetta-run
        sudo chown -R ef /rosetta-run/
        > sudo mkfs -t xfs /dev/sdb
        sudo mount /dev/sdb /rosetta-run
        > - UNMOUNT
            sudo umount -d /dev/sdb

    - Note:
        - if I stop the instance and then restart it, I'll have
          to remount the disk again
