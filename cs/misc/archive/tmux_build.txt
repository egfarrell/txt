### NEW: (CAMK) build without root:
     - LIBEVENT:   
        ai libssl-dev
        wget https://github.com/libevent/libevent/releases/download/release-2.1.12-stable/libevent-2.1.12-stable.tar.gz
        utar libevent-2.1.12-stable.tar.gz
        c libevent-2.1.12-stable
        ./configure --disable-shared --prefix="$HOME/.local"
        make verify

    - NCURSES:  
        wget https://invisible-mirror.net/archives/ncurses/ncurses-6.2.tar.gz
        utar ncurses-6.2.tar.gz


    - TMUX:
        wget https://github.com/tmux/tmux/releases/download/2.9a/tmux-2.9a.tar.gz
        ./configure --prefix="$HOME/.local" \
                CFLAGS="-I$HOME/.local/include" \
                LDFLAGS="-L$HOME/.local/lib"
